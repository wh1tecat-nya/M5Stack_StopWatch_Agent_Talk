cmake_minimum_required(VERSION 3.16)
project(sumi_ui_test C CXX)
get_filename_component(MUSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(SDK "${MUSE_ROOT}/.cache/muse-gadget-sdk-463bd2a05380fea4211b0099ad0f70933c108b20/esp32")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDK}/managed_components/lvgl__lvgl" lvgl)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/fakes")
foreach(header muse_state.h muse_settings.h muse_voice.h muse_link.h muse_wifi.h muse_ble.h esp_timer.h esp_heap_caps.h esp_log.h nvs.h)
    file(WRITE "${CMAKE_BINARY_DIR}/fakes/${header}" "#include \"ui_platform.h\"\n")
endforeach()
add_executable(render_ui render_ui.c "${SDK}/components/muse/sumi_assets.c" "${SDK}/components/muse/sumi_font_24.c" "${SDK}/components/muse/sumi_clock_64.c"
    "${MUSE_ROOT}/overlay/sumi_chat.c" "${MUSE_ROOT}/overlay/sumi_voice_options.c"
    "${SDK}/components/muse/fonts/muse_font_cjk_16.c" "${SDK}/components/muse/fonts/muse_font_latin_16.c")
target_include_directories(render_ui PRIVATE "${CMAKE_BINARY_DIR}/fakes" "${CMAKE_CURRENT_LIST_DIR}" "${SDK}/components/muse")
target_link_libraries(render_ui PRIVATE lvgl m)
